```typescript
export function useNotes() {
  const [notes, setNotes] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  const fetchNotes = useCallback(async () => {
    try {
      setLoading(true);
      const data = await window.api.notes.getAll();
      setNotes(data);
      setError(null);
    } catch (err) {
      setError(err);
    } finally {
      setLoading(false);
    }
  }, []);
  
  useEffect(() => {
    fetchNotes();
  }, [fetchNotes]);
  
  return { notes, loading, error, refetch: fetchNotes };
}
```

```typescript
  const [loading, setLoading] = useState(false);
  
  const createNote = useCallback(async (data: { title: string; body: string }) => {
    setLoading(true);
    try {
      const note = await window.api.notes.create(data);
      return note;
    } finally {
      setLoading(false);
    }
  }, []);
  
  return { createNote, loading };
}
```

```typescript
  const { note, loading } = useNote(noteId);
  const { updateNote } = useUpdateNote();
  const { deleteNote } = useDeleteNote();
  const [title, setTitle] = useState('');
  const [body, setBody] = useState('');
  const editorRef = useRef<EditorView>();
  
  useEffect(() => {
    if (note) {
      setTitle(note.title);
      setBody(note.body);
    }
  }, [note]);
  
  // Auto-save after 2 seconds of inactivity
  useEffect(() => {
    const timer = setTimeout(() => {
      if (note && (title !== note.title || body !== note.body)) {
        updateNote({ id: noteId, title, body });
      }
    }, 2000);
    return () => clearTimeout(timer);
  }, [title, body, note, noteId, updateNote]);
  
  // Initialize CodeMirror
  useEffect(() => {
    const editor = new EditorView({
      doc: body,
      extensions: [
        markdown(),
        inlineHintsPlugin, // Custom plugin for formatting hints
        wikiLinkPlugin,    // Custom plugin for [[wiki-links]]
        EditorView.updateListener.of((update) => {
          if (update.docChanged) {
            setBody(update.state.doc.toString());
          }
        }),

      parent: editorContainerRef.current,
    });
    
    editorRef.current = editor;
    return () => editor.destroy();
```

Create inline hints plugin per RESEARCH.md Pattern 7:
```typescript
const inlineHintsPlugin = ViewPlugin.fromClass(class {
  decorations: DecorationSet;
  
  constructor(view: EditorView) {
    this.decorations = this.buildDecorations(view);
  }
  
  update(update: ViewUpdate) {
    if (update.docChanged || update.viewportChanged) {
      this.decorations = this.buildDecorations(update.view);
    }
  }
  
  buildDecorations(view: EditorView) {
    const decorations: any[] = [];
    
    for (const { from, to } of view.visibleRanges) {
      syntaxTree(view.state).iterate({
        from, to,
        enter: (node) => {
          // Bold: **text**
          if (node.name === 'StrongEmphasis') {
            decorations.push(
              Decoration.mark({ class: 'cm-strong' }).range(node.from + 2, node.to - 2)
            );
          }
          // Italic: *text*
          if (node.name === 'Emphasis') {
            decorations.push(
              Decoration.mark({ class: 'cm-em' }).range(node.from + 1, node.to - 1)
            );
          }
          // Headers: # text
          if (node.name.startsWith('ATXHeading')) {
            decorations.push(
              Decoration.mark({ class: `cm-heading cm-heading-${node.name.slice(-1)}` }).range(node.from, node.to)
            );
          }
        },
      });
    }
    
    return Decoration.set(decorations);
  }
}, {
  decorations: v => v.decorations,
});
```

```typescript
interface TagsInputProps {

export function TagsInput({ noteId }: TagsInputProps) {
  const { tags: allTags } = useTags();
  const [noteTags, setNoteTags] = useState([]);
  
  useEffect(() => {
    async function fetchNoteTags() {
      const data = await window.api.tags.getForNote(noteId);
      setNoteTags(data);
    }
    fetchNoteTags();
  }, [noteId]);
  
  const handleAddTag = async (tagName: string) => {
    await window.api.tags.addToNote(noteId, [tagName]);
    const updated = await window.api.tags.getForNote(noteId);
    setNoteTags(updated);
  };
  
  const handleRemoveTag = async (tagId: string) => {
    await window.api.tags.removeFromNote(noteId, tagId);
    const updated = await window.api.tags.getForNote(noteId);
    setNoteTags(updated);
  };
  
  return (
    <div className="tags-input">
      <Select
        isMulti
        value={noteTags.map(t => ({ value: t.id, label: t.name }))}
        options={allTags.map(t => ({ value: t.id, label: t.name }))}
        onChange={(selected) => {
          const tagNames = selected.map(s => s.label);
          window.api.tags.setForNote(noteId, tagNames).then(() => {
            window.api.tags.getForNote(noteId).then(setNoteTags);
          });
        }}
        placeholder="Add tags..."
      />
    </div>
  );
}
```

```typescript
interface QuickNavProps {
  onNavigate: (noteId: string) => void;
}

export function QuickNav({ onNavigate }: QuickNavProps) {
  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);
  const [selectedIndex, setSelectedIndex] = useState(0);
  
  // Listen for Cmd+K / Ctrl+K
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
        e.preventDefault();
        setOpen(true);
      }
    };
    
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);
  
  // Debounced search
  useEffect(() => {
    if (!query) {
      setResults([]);
      return;
    }
    
    const timer = setTimeout(async () => {
      const data = await window.api.search.quickNav(query);
      setResults(data);
      setSelectedIndex(0);
    }, 200);
    
    return () => clearTimeout(timer);
  }, [query]);
  
  const handleKeyDown = (e: KeyboardEvent) => {
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      setSelectedIndex(i => Math.min(i + 1, results.length - 1));
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      setSelectedIndex(i => Math.max(i - 1, 0));
    } else if (e.key === 'Enter' && results[selectedIndex]) {
      e.preventDefault();
      onNavigate(results[selectedIndex].id);
      setOpen(false);
      setQuery('');
    }
  };
  
  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogContent>
        <input
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          onKeyDown={handleKeyDown}
          placeholder="Search notes..."
          autoFocus
        />
        <ul>
          {results.map((result, index) => (
            <li
              key={result.id}
              className={index === selectedIndex ? 'selected' : ''}
              onClick={() => {
                onNavigate(result.id);
                setOpen(false);
                setQuery('');
              }}
            >
              {result.title}
            </li>
          ))}
        </ul>
      </DialogContent>
    </Dialog>
  );
}
```
