```typescript
  id: string;
  title: string;
  similarity: number;
}
```

**Hook: useSemanticLinks(noteId: string)**
- useState for links (SemanticLink[]), loading (boolean), error (Error | null)
- useCallback for fetchSemanticLinks:
  ```typescript
  const fetchSemanticLinks = useCallback(async () => {
    try {
      setLoading(true);
      const data = await window.api.links.getSemanticLinks(noteId);
      setLinks(data);
      setError(null);
    } catch (err) {
      setError(err as Error);
      setLinks([]);
    } finally {
      setLoading(false);
    }
  }, [noteId]);
  ```
- useEffect to trigger fetchSemanticLinks on mount and when noteId changes
- Return { links, loading, error, refetch: fetchSemanticLinks }

**Extend electron/preload.ts:**

  ```typescript
  }
  ```

  ```typescript
  ipcMain.handle('links:getSemanticLinks', async (event, { noteId }) => {
  ```
