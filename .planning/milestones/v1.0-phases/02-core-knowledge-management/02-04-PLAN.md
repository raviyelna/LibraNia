addTagsToNote implementation:
```typescript
export async function addTagsToNote(noteId: string, tagNames: string[]) {
  const db = getDatabase();
  const tagIds: string[] = [];
  
  // Create or get existing tags
  for (const name of tagNames) {
    const tag = await createTag(name); // Reuses existing if duplicate
    tagIds.push(tag.id);
  }
  
  // Create note-tag associations (ignore duplicates)
  for (const tagId of tagIds) {
    try {
      await db.insert(noteTags).values({ noteId, tagId });
    } catch (error) {
      // Ignore duplicate key errors (tag already associated)
      if (!error.message.includes('UNIQUE constraint')) throw error;
    }
  }
  
  return tagIds;
}
```

removeTagFromNote implementation:
```typescript
export async function removeTagFromNote(noteId: string, tagId: string) {
  const db = getDatabase();
  await db.delete(noteTags).where(
    and(eq(noteTags.noteId, noteId), eq(noteTags.tagId, tagId))
  );
}
```

getNoteTags implementation:
```typescript
export async function getNoteTags(noteId: string) {
  const db = getDatabase();
  const results = await db
    .select({ tag: tags })
    .from(noteTags)
    .innerJoin(tags, eq(noteTags.tagId, tags.id))
    .where(eq(noteTags.noteId, noteId))
    .orderBy(asc(tags.name));
  
  return results.map(r => r.tag);
}
```

getNotesByTag implementation per D-06 (user chose tag-based filtering):
```typescript
export async function getNotesByTag(tagId: string) {
  const db = getDatabase();
  const results = await db
    .select({ note: notes })
    .from(noteTags)
    .innerJoin(notes, eq(noteTags.noteId, notes.id))
    .where(
      and(
        eq(noteTags.tagId, tagId),
        isNull(notes.deletedAt) // Exclude soft-deleted notes
      )
    )
    .orderBy(desc(notes.updatedAt));
  
  return results.map(r => r.note);
}
```

setNoteTags implementation (replace all tags):
```typescript
export async function setNoteTags(noteId: string, tagNames: string[]) {
  const db = getDatabase();
  
  // Delete existing associations
  await db.delete(noteTags).where(eq(noteTags.noteId, noteId));
  
  // Add new associations
  if (tagNames.length > 0) {
    await addTagsToNote(noteId, tagNames);
  }
}
```
