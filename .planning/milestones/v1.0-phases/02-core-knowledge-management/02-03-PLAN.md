export async function quickNavSearch(query: string, limit = 50) {
  const db = getDatabase();
  
  // Escape FTS5 special characters
  const escapedQuery = query.replace(/['"*]/g, '');
  
  // Title-only search with prefix matching
  const results = db.prepare(`
    SELECT 
      n.id,
      n.title,
      n.updated_at,
      fts.rank
    FROM notes_fts fts
    INNER JOIN notes n ON fts.rowid = n.rowid
    WHERE notes_fts MATCH ?
      AND n.deleted_at IS NULL
    ORDER BY fts.rank
    LIMIT ?
  `).all(`title:${escapedQuery}*`, limit);
  
  return results;
}
```

Use better-sqlite3's prepare() for parameterized queries (prevents SQL injection). Use title: prefix in MATCH query to search title column only. Append * for prefix matching ("rea" matches "react", "reading", etc.).

export async function fullTextSearch(query: string, limit = 50) {
  const db = getDatabase();
  
  // Escape FTS5 special characters
  const escapedQuery = query.replace(/['"*]/g, '');
  
  // 7 days ago in milliseconds
  const sevenDaysAgo = Date.now() - 7 * 24 * 60 * 60 * 1000;
  
  const results = db.prepare(`
    SELECT 
      n.id,
      n.title,
      snippet(fts, 1, '<mark>', '</mark>', '...', 32) as snippet,
      n.updated_at,
      fts.rank * (CASE WHEN n.updated_at > ? THEN 1.5 ELSE 1.0 END) as score
    FROM notes_fts fts
    INNER JOIN notes n ON fts.rowid = n.rowid
    WHERE notes_fts MATCH ?
      AND n.deleted_at IS NULL
    ORDER BY score
    LIMIT ?
  `).all(sevenDaysAgo, escapedQuery, limit);
  
  return results;
}
```

Use snippet() function to generate highlighted excerpts from body (column index 1). Recency boost: notes updated in last 7 days get 1.5x multiplier per D-03 (user locked recency boost formula).
